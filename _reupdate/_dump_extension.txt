8	interface ChatSessionInfo { id: string; title: string; }
9	
10	class ChatSessionsProvider implements vscode.TreeDataProvider<ChatSessionInfo> {
11	  private _onDidChangeTreeData = new vscode.EventEmitter<void>();
12	  readonly onDidChangeTreeData = this._onDidChangeTreeData.event;
13	
14	  constructor(private context: vscode.ExtensionContext) {}
15	
16	  refresh(): void { this._onDidChangeTreeData.fire(); }
17	
18	  getTreeItem(session: ChatSessionInfo): vscode.TreeItem {
19	    const activeId = this.context.globalState.get<string>('amiAiClaw.activeSessionId', 'default');
20	    const isActive = session.id === activeId;
21	    const item = new vscode.TreeItem(session.title, vscode.TreeItemCollapsibleState.None);
22	    item.command = { command: 'amiAiClaw.switchChatSession', title: '切換到此聊天', arguments: [session.id] };
23	    item.iconPath = new vscode.ThemeIcon(isActive ? 'comment-discussion' : 'comment');
24	    item.contextValue = 'chatSession';
25	    item.tooltip = session.title;
26	    if (isActive) { item.description = '▶ 活躍'; }
27	    return item;
28	  }
29	
30	  getChildren(): vscode.ProviderResult<ChatSessionInfo[]> {
31	    const sessions = this.context.globalState.get<ChatSessionInfo[]>('amiAiClaw.sessions', []);
32	    return sessions.length > 0 ? sessions : [{ id: 'default', title: '聊天 1' }];
33	  }
34	}

36	function openAndSend(context: vscode.ExtensionContext, msg: object) {
37	  // createOrShow 為 async（內部會 await sidebar.focus），需先等它完成再 post
38	  void (async () => {
39	    try {
40	      await OllamaChatPanel.createOrShow(context);
41	      const panel = OllamaChatPanel.currentPanel;
42	      if (panel) {
43	        await panel.waitForWebviewReady();
44	        panel.postMessageToWebview(msg);
45	      }
46	    } catch (e) {
47	      OllamaChatPanel.reportDiagnostic('openAndSend failed', e);
48	      OllamaChatPanel.revealDiagnostics();
49	      void vscode.window.showErrorMessage('AmiClaw 無法開啟，請查看輸出視窗「AmiClaw Diagnostics」。');
50	    }
51	  })();
52	}

97	  // ── 對話記錄回呼：Agent 完成後將提示 + 回應 append 到 AmiClawTodoLog.md ────
98	  OllamaChatPanel.onTodoComplete = async (prompt: string, response: string) => {
99	    const folders = vscode.workspace.workspaceFolders ?? [];
100	    // 找到 AmiClawToClaudeToDo.md 所在的 folder，把 log 放在同目錄
101	    let logFolder = folders[0]?.uri;
102	    for (const folder of folders) {
103	      try {
104	        await vscode.workspace.fs.stat(vscode.Uri.joinPath(folder.uri, 'AmiClawToClaudeToDo.md'));
105	        logFolder = folder.uri;
106	        break;
107	      } catch { /* 不在這個 folder */ }
108	    }
109	    if (!logFolder) { return; }
110	    const logUri = vscode.Uri.joinPath(logFolder, 'AmiClawTodoLog.md');
111	    const ts = new Date().toISOString().replace('T', ' ').replace(/\.\d+Z$/, ' UTC');
112	    const entry = `\n---\n## 📋 ${ts} 任務\n${prompt}\n\n## 🤖 ${ts} Agent 回應\n${response}\n`;
113	    try {
114	      // 追加到已有的 log（如果存在），否則建立新檔
115	      let existing = '';
116	      try {
117	        const b = await vscode.workspace.fs.readFile(logUri);
118	        existing = Buffer.from(b).toString('utf8');
119	      } catch { /* 第一次建立 */ }
120	      const newContent = Buffer.from(existing + entry, 'utf8');
121	      await vscode.workspace.fs.writeFile(logUri, newContent);
122	      todoLog.appendLine(`[onTodoComplete] 已寫入 ${logUri.fsPath}（+${entry.length} 字）`);
123	      vscode.window.setStatusBarMessage(`📝 AmiClaw：對話已記錄到 AmiClawTodoLog.md`, 5000);
124	    } catch (e) {
125	      todoLog.appendLine(`[onTodoComplete] 寫入失敗: ${(e as Error)?.message ?? e}`);
126	    }
127	  };

129	  let _todoDebounce: ReturnType<typeof setTimeout> | undefined;
130	
131	  async function processTodoFile(uri: vscode.Uri) {
132	    clearTimeout(_todoDebounce);
133	    _todoDebounce = setTimeout(async () => {
134	      try {
135	        todoLog.appendLine(`[processTodoFile] 觸發: ${uri.fsPath}`);
136	        const bytes = await vscode.workspace.fs.readFile(uri);
137	        const prompt = Buffer.from(bytes).toString('utf8').trim();
138	        if (!prompt) {
139	          todoLog.appendLine('[processTodoFile] 檔案為空，跳過');
140	          return;
141	        }
142	        todoLog.appendLine(`[processTodoFile] 內容長度=${prompt.length}，送入 agentSend`);
143	        // 確保 panel 已建立，再進入 todo 模式
144	        await OllamaChatPanel.createOrShow(context);
145	        OllamaChatPanel.currentPanel?.enterTodoMode(prompt);
146	        const send = () => OllamaChatPanel.currentPanel?.postMessageToWebview({ type: 'externalAgentSend', prompt });
147	        if (OllamaChatPanel.currentPanel) { send(); } else { setTimeout(send, 700); }
148	        vscode.window.setStatusBarMessage(`🤖 AmiClaw：已讀取 AmiClawToClaudeToDo.md (${prompt.length} 字)`, 5000);
149	      } catch (e) {
150	        const msg = (e as Error)?.message ?? String(e);
151	        todoLog.appendLine(`[processTodoFile] 錯誤: ${msg}`);
152	        vscode.window.showErrorMessage(`AmiClaw 讀取 AmiClawToClaudeToDo.md 失敗：${msg}`);
153	      }
154	    }, 500);
155	  }

438	          const org = await vscode.window.showInputBox({ prompt: '原始資料夾路徑 (ORG)', placeHolder: '例：MyFeature/ORG' });
439	          if (!org) { return; }
440	          target = `by taking the diff between the ${mod.trim()} and ${org.trim()} folders`;
441	          break;
442	        }
443	        default: return;
444	      }
445	
446	      // SourceTag
447	      const tagOptions = [
448	        { label: 'AptioV (預設)', description: 'AMI Aptio V UEFI BIOS', id: 'AptioV' },
449	        { label: 'AptioV + Unittesting', description: 'AptioV 含 Unit Test', id: 'AptioV, Unittesting' },
450	        { label: 'AptioPE', description: '純 UEFI/EDK2，無 AMI 建置系統', id: 'AptioPE' },
451	        { label: 'AMI Porting Changes Only', description: '只看 AMI porting 部分', id: '' },
452	      ];
453	      const tagPick = await vscode.window.showQuickPick(tagOptions, { title: 'SourceTag', placeHolder: '選擇 review 規則集…' });
454	      if (!tagPick) { return; }
455	
456	      // Jira ID（可選）
457	      const jiraId = await vscode.window.showInputBox({ prompt: 'Jira ID（可留空）', placeHolder: '例：AOC-1234' }) ?? '';
458	
459	      // 組合 prompt
460	      const sourceTagLine = tagPick.id ? `\nSourceTag: ${tagPick.id}` : '';
461	      const jiraLine = jiraId.trim() ? `\nJira Id: ${jiraId.trim()}` : '';
462	      const amiPortingLine = tagPick.id === '' ? ' AMI porting changes only' : '';
463	      const prompt = `Perform a code review of ${target}${amiPortingLine}.${sourceTagLine}${jiraLine}`;
464	
465	      // externalAgentSend 會在 webview 側先呼叫 setInteractionMode('agent')，確保從 Team 切回 Agent
466	      openAndSend(context, { type: 'externalAgentSend', prompt });
467	    })
468	  );

522	      try {
523	        if (!vscode.workspace.workspaceFolders || vscode.workspace.workspaceFolders.length === 0) {
524	          vscode.window.showInformationMessage('No workspace folder is open.');
525	          return;

533	        chan.appendLine(msg);

542	  context.subscriptions.push(
543	    vscode.commands.registerCommand('amiAiClaw.execFileCountShell', async () => {
544	      try {
545	        if (!vscode.workspace.workspaceFolders || vscode.workspace.workspaceFolders.length === 0) {
546	          vscode.window.showInformationMessage('No workspace folder is open.');
547	          return;
548	        }
549	        const activeUri = vscode.window.activeTextEditor?.document.uri;
550	        const wf = (activeUri && vscode.workspace.getWorkspaceFolder(activeUri)) ?? vscode.workspace.workspaceFolders[0];
551	        const root = wf.uri.fsPath;
552	        let cmd: string;
553	        if (process.platform === 'win32') {
554	          // Use PowerShell for reliable recursive file count
555	          cmd = `powershell -NoProfile -Command "(Get-ChildItem -Path '${root.replace(/'/g, "''")}' -File -Recurse -ErrorAction SilentlyContinue | Measure-Object).Count"`;
556	        } else {
557	          // POSIX: use ls -R | wc -l
558	          cmd = `sh -c "ls -R '${root.replace(/'/g, "'\\''")}' | wc -l"`;
559	        }
560	
561	        const oc = vscode.window.createOutputChannel('AmiClaw');
562	        oc.appendLine(`Running: ${cmd}`);
563	        oc.show(true);
564	        try {
565	          const { stdout, stderr } = await exec(cmd, { windowsHide: true, maxBuffer: 50 * 1024 * 1024 });
566	          if (stderr && String(stderr).trim()) oc.appendLine(String(stderr));
567	          const n = parseInt(String(stdout).trim().split(/\r?\n/).pop() || '', 10);
568	          if (!Number.isNaN(n)) {
569	            const msg = `Shell file count: ${n.toLocaleString()} (path: ${root})`;
570	            oc.appendLine(msg);
571	            vscode.window.showInformationMessage(msg);
572	            return;
573	          }
574	        } catch (e) {
575	          oc.appendLine('Shell command failed: ' + (e instanceof Error ? e.message : String(e)));
576	        }
577	
578	        // Fallback to JS counting via workspace.findFiles
579	        oc.appendLine('Falling back to workspace.findFiles count...');
580	        const exclude = '{**/node_modules/**,**/.git/**,**/dist/**,**/out/**}';
581	        const uris = await vscode.workspace.findFiles('**/*', exclude);
582	        const count = uris.length;
583	        const msg = `Fallback file count: ${count.toLocaleString()} (excludes node_modules, .git, dist, out)`;
584	        oc.appendLine(msg);
585	        vscode.window.showInformationMessage(msg);
586	      } catch (e) {
587	        vscode.window.showErrorMessage('Failed to execute file-count: ' + (e instanceof Error ? e.message : String(e)));
588	      }
589	    })
590	  );
591	}
