 130|   // 建立一個 Proxy 用以攔截陣列變更並標記為需要儲存
 131|   private createHistoryProxy(arr: ChatMessage[], sessionId: string): ChatMessage[] {
 132|     const that = this;
 133|     const target = Array.isArray(arr) ? arr : [];
 134|     const handler: ProxyHandler<ChatMessage[]> = {
 135|       set(t, prop, value) {
 136|         const res = Reflect.set(t, prop, value);
 137|         // 設定長度或數字索引時視為修改
 138|         if (typeof prop === 'string' && (prop === 'length' || /^\d+$/.test(prop))) {
 139|           that._chatDirty = true;
 140|           that.schedulePersistChatHistories();
 141|         }
 142|         return res;
 143|       },
 144|       get(t, prop, receiver) {
 145|         const v = Reflect.get(t, prop, receiver);
 146|         if (typeof v === 'function') {
 147|           const name = String(prop);
 148|           if (['push', 'pop', 'shift', 'unshift', 'splice', 'sort', 'reverse', 'fill', 'copyWithin'].includes(name)) {
 149|             return function (...args: any[]) {
 150|               const r = (Array.prototype as any)[name].apply(t, args);
 151|               that._chatDirty = true;
 152|               that.schedulePersistChatHistories();
 153|               return r;
 154|             };
 155|           }
 156|         }
 157|         return v;
 158|       }
 159|     };
 160|     const proxy = new Proxy(target, handler) as ChatMessage[];
 161|     (proxy as any).__isProxy = true;
 162|     return proxy;
 163|   }

 166|   private wrapAllHistories(): void {
 167|     for (const [id, arr] of Object.entries(this._chatHistories)) {
 168|       if (!arr || !(arr as any).__isProxy) {
 169|         this._chatHistories[id] = this.createHistoryProxy(arr || [], id);

 174|       this._chatHistories[this._activeSessionId] = this.createHistoryProxy([], this._activeSessionId);
 175|     }

 184|   private async persistChatHistories(): Promise<void> {
 185|     if (!this._chatDirty) return;
 186|     try {
 187|       const copy: Record<string, ChatMessage[]> = {};
 188|       for (const [sid, arr] of Object.entries(this._chatHistories)) {
 189|         const msgs = Array.isArray(arr) ? arr.slice(-this._maxMessagesPerSession) : [];
 190|         copy[sid] = msgs.map(m => ({ role: m.role, content: (m.content ?? '').slice(0, 10000) }));
 191|       }
 192|       // 改用 workspaceState 实現工作區隔離
 193|       await this._context.workspaceState.update('amiAiClaw.chatHistories', copy);
 194|       this._chatDirty = false;
 195|     } catch (e) {
 196|       OllamaChatPanel.log('persistChatHistories error: ' + (e instanceof Error ? e.message : String(e)));
 197|     }

 230|   /** 記錄一次 API 呼叫的 token 使用量，並推送更新到前端。 */
 231|   private trackUsage(model: string, tokens: number, multiplier = '', toolCall = false): void {
 232|     if (!toolCall && (!tokens || tokens <= 0)) { return; }
 233|     const isCopilot = model.startsWith('copilot::') || model.startsWith('copilot/');
 234|     const key = model.replace(/^copilot[::\/]+/, '');
 235|     const existing = this._usageStats[key];
 236|     if (existing) {
 237|       existing.tokens += tokens;
 238|       if (tokens > 0) existing.calls = (existing.calls || 0) + 1;
 239|       if (toolCall) existing.toolCalls = (existing.toolCalls || 0) + 1;
 240|     } else {
 241|       this._usageStats[key] = { tokens, isCopilot, multiplier, calls: tokens > 0 ? 1 : 0, toolCalls: toolCall ? 1 : 0 };
 242|     }
 243|     // 持久化累計值
 244|     const saved = this._context.globalState.get<Record<string, { tokens: number; isCopilot: boolean; multiplier: string; calls: nu
 245|     const sk = saved[key];
 246|     if (sk) { sk.tokens += tokens; if (tokens > 0) sk.calls = (sk.calls || 0) + 1; if (toolCall) sk.toolCalls = (sk.toolCalls || 0
 247|     else { saved[key] = { tokens, isCopilot, multiplier, calls: tokens > 0 ? 1 : 0, toolCalls: toolCall ? 1 : 0 }; }
 248|     this._context.globalState.update('amiAiClaw.usageStats', saved);
 249|     this._panel.webview.postMessage({ type: 'usageUpdate', stats: this._usageStats });
 250|   }
